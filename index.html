Google Apps Script Web App URL

เมื่อเชื่อมต่อแล้ว การกดอัปโหลดเกม แก้ไขข้อมูลเกม ลบเกม หรือนับยอดเข้าเล่น จะถูกบันทึกลงใน Google Sheets ทันที

โค้ดสำหรับวางใน Google Apps Script (Code.gs) — รองรับการเพิ่ม/แก้ไข/ลบข้อมูล 

const SHEET_NAME = 'Games';
const HEADERS = ['id', 'title', 'subject', 'grade', 'description', 'gameUrl', 'image', 'author', 'views', 'createdAt'];

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(HEADERS);
    sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold').setBackground('#e0e7ff');
  } else if (sheet.getLastRow() === 0) {
    sheet.appendRow(HEADERS);
    sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold').setBackground('#e0e7ff');
  }
  return sheet;
}

function doGet(e) {
  try {
    const sheet = getOrCreateSheet();
    const rows = sheet.getDataRange().getValues();
    if (rows.length <= 1) {
      return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: [] }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    const headers = rows[0];
    const data = rows.slice(1).map(row => {
      const obj = {};
      headers.forEach((h, i) => obj[h] = row[i]);
      return obj;
    }).filter(item => item.id && item.title);

    return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: data.reverse() }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    const sheet = getOrCreateSheet();
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action || 'add';

    if (action === 'add') {
      const game = payload.game || payload;
      const newRow = HEADERS.map(h => {
        if (h === 'id') return game.id || 'game_' + new Date().getTime();
        if (h === 'views') return Number(game.views || game.plays || 0);
        if (h === 'createdAt') return game.createdAt || new Date().toISOString().split('T')[0];
        if (h === 'image') return game.image || game.imageUrl || '';
        if (h === 'gameUrl') return game.gameUrl || game.playUrl || '';
        return game[h] !== undefined ? game[h] : '';
      });
      sheet.appendRow(newRow);
      return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Added' }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'edit') {
      const game = payload.game || payload;
      const rows = sheet.getDataRange().getValues();
      let updated = false;
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === String(game.id)) {
          const updatedRow = HEADERS.map((h, colIdx) => {
            if (h === 'id') return game.id;
            if (h === 'views') return Number(game.views ?? game.plays ?? rows[i][colIdx] ?? 0);
            if (h === 'createdAt') return game.createdAt || rows[i][colIdx] || new Date().toISOString().split('T')[0];
            if (h === 'image') return game.image || game.imageUrl || rows[i][colIdx] || '';
            if (h === 'gameUrl') return game.gameUrl || game.playUrl || rows[i][colIdx] || '';
            return game[h] !== undefined ? game[h] : rows[i][colIdx];
          });
          sheet.getRange(i + 1, 1, 1, HEADERS.length).setValues([updatedRow]);
          updated = true;
          break;
        }
      }
      if (!updated) {
        const newRow = HEADERS.map(h => {
          if (h === 'id') return game.id || 'game_' + new Date().getTime();
          if (h === 'views') return Number(game.views || game.plays || 0);
          if (h === 'createdAt') return game.createdAt || new Date().toISOString().split('T')[0];
          if (h === 'image') return game.image || game.imageUrl || '';
          if (h === 'gameUrl') return game.gameUrl || game.playUrl || '';
          return game[h] !== undefined ? game[h] : '';
        });
        sheet.appendRow(newRow);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Updated' }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'delete' && payload.id) {
      const rows = sheet.getDataRange().getValues();
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === String(payload.id)) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Deleted' }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'incrementView' && payload.id) {
      const rows = sheet.getDataRange().getValues();
      const viewColIndex = HEADERS.indexOf('views') + 1;
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === String(payload.id)) {
          const currentViews = Number(rows[i][viewColIndex - 1] || 0);
          sheet.getRange(i + 1, viewColIndex).setValue(currentViews + 1);
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'View updated' }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Unknown action' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
